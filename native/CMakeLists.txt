cmake_minimum_required(VERSION 3.22)
project(PRTSNative VERSION 0.2 LANGUAGES C CXX)
set(PRTS_VLM_MICROBATCH 256 CACHE STRING "Language prefill microbatch, compare quality before changing")
set(PRTS_LIBRARY_NAME prts_vlm CACHE STRING "Output name; permits preserving reference binaries")
option(PRTS_OFFICIAL_RUNTIME "Use the pinned official llama.cpp b10964 APIs and structured sampling" OFF)
option(PRTS_CPU_REPACK "Allow additional packed CPU copies of quantized weights" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED LLAMA_SOURCE_DIR)
    message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned llama.cpp-omni checkout; see native/README.md")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL ${APPLE} CACHE BOOL "")
add_subdirectory(${LLAMA_SOURCE_DIR} llama)
set(LLAMA_INSTALL_VERSION 0.0.0)
add_subdirectory(${LLAMA_SOURCE_DIR}/tools/mtmd mtmd)
add_library(prts_vlm SHARED prts_vlm.cpp portable/prts_geometry.cpp portable/prts_onnx.cpp portable/prts_image.cpp portable/prts_queue.cpp)
target_include_directories(prts_vlm PUBLIC include portable/include)
target_include_directories(prts_vlm PRIVATE third_party/onnxruntime)
target_compile_definitions(prts_vlm PRIVATE PRTS_GEOMETRY_BUILD)
target_compile_definitions(prts_vlm PRIVATE PRTS_VLM_MICROBATCH=${PRTS_VLM_MICROBATCH})
if(PRTS_OFFICIAL_RUNTIME)
    target_compile_definitions(prts_vlm PRIVATE PRTS_OFFICIAL_RUNTIME)
endif()
if(PRTS_CPU_REPACK)
    target_compile_definitions(prts_vlm PRIVATE PRTS_CPU_REPACK_ENABLED)
endif()
set_target_properties(prts_vlm PROPERTIES OUTPUT_NAME ${PRTS_LIBRARY_NAME})
target_link_libraries(prts_vlm PRIVATE llama mtmd llama-common)
if(APPLE)
    set_target_properties(prts_vlm PROPERTIES
        FRAMEWORK TRUE
        FRAMEWORK_VERSION A
        MACOSX_FRAMEWORK_IDENTIFIER org.prts.core.vlm
        PUBLIC_HEADER "include/prts_vlm.h;portable/include/prts_geometry.h;portable/include/prts_onnx.h;portable/include/prts_image.h;portable/include/prts_queue.h"
        XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
    add_custom_command(TARGET prts_vlm POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:prts_vlm>/Modules"
        COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/module.modulemap"
                "$<TARGET_BUNDLE_DIR:prts_vlm>/Modules/module.modulemap")
endif()
install(TARGETS prts_vlm LIBRARY DESTINATION lib RUNTIME DESTINATION bin
    FRAMEWORK DESTINATION Frameworks PUBLIC_HEADER DESTINATION include)
install(FILES include/prts_vlm.h DESTINATION include)
install(FILES portable/include/prts_geometry.h DESTINATION include)
install(FILES portable/include/prts_onnx.h DESTINATION include)
install(FILES portable/include/prts_image.h DESTINATION include)
install(FILES portable/include/prts_queue.h DESTINATION include)

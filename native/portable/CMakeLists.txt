cmake_minimum_required(VERSION 3.22)
project(PRTSPortable LANGUAGES CXX)
foreach(module geometry image onnx queue)
    add_library(prts_${module} SHARED prts_${module}.cpp)
    target_compile_features(prts_${module} PRIVATE cxx_std_17)
    target_include_directories(prts_${module} PUBLIC include)
    set_target_properties(prts_${module} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
target_compile_definitions(prts_geometry PRIVATE PRTS_GEOMETRY_BUILD)
target_include_directories(prts_onnx PRIVATE ../third_party/onnxruntime)
